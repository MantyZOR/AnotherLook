if(BUILD_TESTING)
# UI probes share the exact UI object files, but are never linked into the app.
add_executable(goodview_ui_tests WIN32 src/main.cpp resources/app.rc
  tests/panel_smoke.cpp tests/interaction_smoke.cpp tests/tray_smoke.cpp
  tests/typography_smoke.cpp tests/desktop_regression.cpp)
target_compile_definitions(goodview_ui_tests PRIVATE GOODVIEW_TESTING=1)
target_link_libraries(goodview_ui_tests PRIVATE goodview_ui)
target_link_options(goodview_ui_tests PRIVATE "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest")
function(goodview_ui_test name mode label)
  add_test(NAME ${name} COMMAND ${CMAKE_COMMAND}
    "-DTEST_APP=$<TARGET_FILE:goodview_ui_tests>" "-DTEST_MODE=${mode}"
    "-DTEST_NAME=${name}" "-DTEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}/test-profiles"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_ui.cmake")
  set_tests_properties(${name} PROPERTIES LABELS "${label}" TIMEOUT 45 RUN_SERIAL TRUE)
endfunction()
